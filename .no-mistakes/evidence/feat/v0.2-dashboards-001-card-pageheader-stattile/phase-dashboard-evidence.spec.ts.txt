import { test, expect } from '@playwright/test';
const evidence = '~/.no-mistakes/evidence/01M7SMMD2XH955PSB96AAGBXGM';
for (const locale of ['en','ar']) for (const theme of ['light','dark']) {
 test(`dashboard fixture composition and captures ${locale} ${theme}`, async ({page}) => {
  for (const name of ['Card','PageHeader','StatTile']) {
   await page.goto(`/components/${name}?locale=${locale}&dir=${locale==='ar'?'rtl':'ltr'}&theme=${theme}`);
   await page.evaluate(()=>document.fonts.ready);
   const fixture=page.getByRole('region',{name:locale==='en'?'OpenUI fixture':'عينة OpenUI'});
   await expect(fixture).toBeVisible();
   for (const width of [1440,390]) {
    await page.setViewportSize({width,height:900});
    await page.screenshot({path:`${evidence}/${name}-${locale}-${theme}-${width}.png`,fullPage:true,animations:'disabled'});
    if(name==='Card') {
     const grid=fixture.locator('.ts-stat-grid');
     await expect(grid.locator('[data-slot="stat-tile"]')).toHaveCount(2);
     await expect(grid).toHaveCSS('display','grid');
     expect(await grid.evaluate(el=>getComputedStyle(el).gridTemplateColumns.split(' ').length)).toBe(width===1440?2:1);
    } else if(name==='StatTile') {
     const img=fixture.getByRole('img',{name:locale==='en'?'Revenue':'الإيرادات',exact:true});
     await expect(img).toBeVisible();
     await expect(img).toHaveCSS('transform',locale==='en'?'none':'matrix(-1, 0, 0, 1, 0, 0)');
    } else {
     await expect(fixture.getByRole('heading',{name:locale==='en'?'Overview':'نظرة عامة',exact:true})).toBeVisible();
     const button=page.locator('.dashboard-component-preview').getByRole('button',{name:locale==='en'?'Export report':'تصدير التقرير'});
     await button.press('Tab');
     await button.focus();
     await expect(button).toBeFocused();
     await button.press('Enter');
    }
    expect(await page.evaluate(()=>document.documentElement.scrollWidth<=innerWidth)).toBe(true);
   }
  }
 });
}
